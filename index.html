Python data visualisation

This Python script imports quarterly return data to compare the performance of an analysed fund against the five top-performing cybersecurity ETFs. The output is a line chart that highlights relative trends over time.

Interactive Python NotebookExplore the code, run the pipeline, and view real-time results via Google Colab. colab.research.google.com
ETF_Comparison.py
import pandas as pd
import matplotlib.pyplot as plt

# Create DataFrame
df = pd.DataFrame({'Quarter': ['Q1 '22', 'Q2 '22', 'Q3 '22', 'Q4 '22', 'Q1 '23', 'Q2 '23', 'Q3 '23', 'Q4 '23', 'Q1 '24', 'Q2 '24', 'Q3 '24', 'Q4 '24', 'Q1 '25'],
    'Analyzed Fund': [-4.88, -22.66, -5.68, -0.74, 11.86, 4.79, 1.62, 17.63, 5.84, -1.07, 4.73, 5.16, -3.94],
    'L&G Cyber Security UCITS ETF': [0.95, -20.25, 6.26, -15.38, 11.81, 6.00, 3.86, 10.22, 4.99, -2.16, 2.14, 20.08, -9.69],
    'First Trust NASDAQ Cybersecurity ETF': [1.02, -24.91, -2.04, -2.47, 10.49, 7.26, 0.39, 18.38, 3.86, -0.54, 5.76, 8.88, -5.72],
    'WisdomTree Cybersecurity UCITS ETF': [-5.81, -28.48, -1.98, -14.08, 21.18, 6.66, 5.54, 24.58, 0.28, -2.26, 0.53, 13.30, -6.65],
    'Rize Cybersecurity and Data Privacy ETF': [-5.51, -22.35, -0.81, -8.27, 12.46, 4.75, 1.73, 22.68, 1.68, -4.26, 3.57, 8.71, -1.01],
    'iShares Digital Security UCITS ETF': [-10.12, -20.18, -3.07, 1.35, 10.48, 5.37, 0.45, 13.94, 3.05, -1.60, 8.96, 4.88, -5.99]
})
pd.set_option('display.width', 300)
df.set_index('Quarter', inplace=True)
print(df.head())

plt.figure(figsize=(13, 5))
for col in df.columns:
    if col == 'Analyzed Fund':
      plt.plot(df.index, df[col], linewidth=2.5, label=col, zorder=3)
      lower_line = df[col] - 1
      plt.fill_between(df.index, lower_line, df[col], alpha=0.2, zorder=2)
    else:
      plt.plot(df.index, df[col], linewidth=0.5, label=col, zorder=1)
plt.legend()
plt.title('Yearly Performance with Quarterly Detail')
plt.xlabel('Quarter')
plt.xticks(rotation=45)
plt.ylabel('Performance (%)')
plt.show()
